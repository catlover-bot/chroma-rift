function tape(area){
 const pose=area==='04'?{position:{x:0,y:1.6,z:-2.5},yaw:Math.PI,pitch:0}:{position:{x:-3.75,y:1.6,z:10},yaw:Math.PI/4,pitch:Math.atan2(-.2,Math.SQRT2)};

  phase='mirror';press('mirror-corridor-mirror',[-2.65,1.9,11.4]);wait(.6);

  phase='key-new-position';walk(-3.75,11.6);press('departure-key',[-3.55,1.12,12.95]);
  phase='procedure';press('departure-procedure',[-4.75,1.4,10]);press('departure-procedure',[-3.15,1.12,13.08]);
  phase='call';press('departure-bell',[-4.75,1.4,11]);press('departure-bell',[-2.72,1.18,12.72]);turn(-2.15,0);wait(5.5);
  phase='isolation';press('departure-door',[-4.75,1.4,12]);press('departure-door',[-2.72,1.18,13.18]);turn(-2.15,0);wait(2);
  phase='power-stop';press('departure-stop',[-4.75,1.4,13]);press('departure-stop',[-3.18,1.55,13.46]);turn(-2.15,0);wait(2);
  phase='staff-exit';press('departure-staff-door',[-3.7,1.5,13.42]);turn(Math.PI,0);wait(1.5);

  phase='aftermath';walk(-3.75,25);wait(3);
 }
 return {schemaVersion:1,area,initialPose:initial,settings:{quality:'standard',horrorIntensity:'subdued',width:390,height:844,verticalFov:65,fps:5},boundary:'Fixed authored operation tape shared byte-for-byte by both revisions; attempts at both old and new control locations are intentional. Refused commands are retained. Initial validated checkpoint only; no mid-run pose/progress injection or conditional route adaptation.',operations:ops};
}
function recording(){const events=[],players=[],states=[];let time=0,seq=0,maxPlayers=0;const log=(operation,details={})=>events.push({sequence:++seq,time,area:'comparison',simulationSeconds:time,operation,...details});
 const backend={availability:'available',prepare:async()=>undefined,createPlayer(source){const id=players.length;let volume=0,loop=false,released=false;const p={get isLoaded(){return true;},get volume(){return volume;},set volume(value){volume=value;log('volume',{id,source,value});},get loop(){return loop;},set loop(value){loop=value;log('loop',{id,source,value});},play(){if(released)throw Error('released');log('play',{id,source});},pause(){log('pause',{id,source});},seekTo(value){log('seek',{id,source,value});return{then(cb){cb();return{catch(){}};}};},release(){if(released)throw Error('duplicate release');released=true;log('release',{id,source});},get released(){return released;}};players.push(p);maxPlayers=Math.max(maxPlayers,players.filter(p=>!p.released).length);log('create',{id,source});return p;}};
 return{backend,advance(dt){time+=dt;},state(owner,actor){const state=owner.getDiagnostics().musicState??'baseline-no-music';if(states.at(-1)?.state!==state)states.push({time,state,actorPhase:actor});},report(){return{durationSeconds:time,maxPlayers,activePlayersAtEnd:players.filter(p=>!p.released).length,events,states,boundary:'Real audio owner driven by the fixed comparison controller tape. Native successful presentation and already-loaded zero-latency seek are explicit fixtures. No natural campaign completion or native audio capture is claimed.'};}};
}
